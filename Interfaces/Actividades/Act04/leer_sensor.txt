#include <Wire.h>

#define MPU6050_ADDR 0x68
#define SDA_PIN 21
#define SCL_PIN 22

void setup() {
  Serial.begin(115200);
  Wire.begin(SDA_PIN, SCL_PIN);

  // Despertar el MPU6050
  Wire.beginTransmission(MPU6050_ADDR);
  Wire.write(0x6B); // Registro PWR_MGMT_1
  Wire.write(0x00); // Activar sensor
  Wire.endTransmission(true);

  Serial.println("MPU6050 inicializado");
}

void loop() {
  // Leer acelerómetro X y Y (registros 0x3B al 0x3E)
  Wire.beginTransmission(MPU6050_ADDR);
  Wire.write(0x3B); // ACCEL_XOUT_H
  Wire.endTransmission(false);
  Wire.requestFrom(MPU6050_ADDR, 4, true); // 4 bytes: X_H, X_L, Y_H, Y_L

  int16_t rawX = (Wire.read() << 8) | Wire.read();
  int16_t rawY = (Wire.read() << 8) | Wire.read();

  // Convertir a g (escala ±2g por defecto → divisor 16384)
  float accelX = rawX / 16384.0;
  float accelY = rawY / 16384.0;

  Serial.printf("Aceleración X: %.3f g  |  Y: %.3f g\n", accelX, accelY);
  delay(8);
}
