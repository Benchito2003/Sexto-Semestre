// Obtención de los datos del acelerometro
enum Eje { EJE_X, EJE_Y, EJE_Z };

float leerEje(Eje eje) {
  Wire.beginTransmission(MPU6050_ADDR);
  Wire.write(0x3B);              // siempre arranca desde ACCEL_XOUT_H
  Wire.endTransmission(false);
  Wire.requestFrom(MPU6050_ADDR, 6, true);

  int16_t raw[3];
  for (int i = 0; i < 3; i++) {
    raw[i] = (Wire.read() << 8) | Wire.read();
  }

  return raw[eje] / 16384.0f;
}
