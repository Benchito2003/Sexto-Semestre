#include "HX711.h"

const int LOADCELL_DOUT_PIN = 16;
const int LOADCELL_SCK_PIN = 4;

HX711 scale;

// Iniciamos con un factor genérico. 
// Para celdas grandes, este número suele rondar entre los 10,000 y 50,000 (positivo o negativo).
float factor_calibracion = 10000.0; 

void setup() {
  Serial.begin(115200);
  Serial.println("--- Modo de Calibración HX711 ---");
  Serial.println("1. Asegúrate de que no haya NADA sobre la báscula.");
  Serial.println("2. Esperando para hacer la tara...");
  delay(2000);
  
  scale.begin(LOADCELL_DOUT_PIN, LOADCELL_SCK_PIN);
  
  // Resetea la escala a 0
  scale.set_scale();
  scale.tare(); 
  
  Serial.println("Tara completada. El peso actual debería ser 0.");
  Serial.println("-------------------------------------------------");
  Serial.println("3. Coloca un peso CONOCIDO sobre la báscula (ej. tu propio peso, garrafones de agua, etc).");
  Serial.println("4. Usa las siguientes teclas para ajustar el factor hasta que el 'Peso' coincida con la realidad:");
  Serial.println("   'a' (+10)    | 'z' (-10)");
  Serial.println("   's' (+100)   | 'x' (-100)");
  Serial.println("   'd' (+1000)  | 'c' (-1000)");
  Serial.println("   'f' (+10000) | 'v' (-10000)");
  Serial.println("-------------------------------------------------\n");
}

void loop() {
  // Aplicar el factor actual
  scale.set_scale(factor_calibracion);
  
  Serial.print("Peso detectado: ");
  // Imprime el peso promediando 10 lecturas para mayor estabilidad
  Serial.print(scale.get_units(10), 2); 
  Serial.print(" kg  |  Factor de calibración actual: ");
  Serial.println(factor_calibracion);

  // Leer comandos del puerto serie
  if (Serial.available()) {
    char temp = Serial.read();
    
    // Ignoramos los saltos de línea (CR/LF) por si la terminal los envía automáticamente
    if (temp == '\n' || temp == '\r') {
      return; 
    }
    
    if (temp == 'a') factor_calibracion += 10;
    else if (temp == 'z') factor_calibracion -= 10;
    else if (temp == 's') factor_calibracion += 100;
    else if (temp == 'x') factor_calibracion -= 100;
    else if (temp == 'd') factor_calibracion += 1000;
    else if (temp == 'c') factor_calibracion -= 1000;
    else if (temp == 'f') factor_calibracion += 10000;
    else if (temp == 'v') factor_calibracion -= 10000;
  }
  
  delay(500);
}
